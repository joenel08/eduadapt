DROP TABLE IF EXISTS `teacher_class_assignments`;

CREATE TABLE `teacher_class_assignments` (
    `id` BIGINT(20) UNSIGNED NOT NULL AUTO_INCREMENT,
    `teacher_profile_id` BIGINT(20) UNSIGNED NOT NULL,
    `class_id` BIGINT(20) UNSIGNED NOT NULL,
    `subject_id` BIGINT(20) UNSIGNED NULL DEFAULT NULL,
    `created_at` TIMESTAMP NULL DEFAULT NULL,
    `updated_at` TIMESTAMP NULL DEFAULT NULL,
    PRIMARY KEY (`id`) USING BTREE,
    UNIQUE INDEX `teacher_class_subject_unique` (`teacher_profile_id`, `class_id`, `subject_id`) USING BTREE,
    INDEX `tca_teacher_profile_id_index` (`teacher_profile_id`) USING BTREE,
    INDEX `teacher_class_assignments_class_id_foreign` (`class_id`) USING BTREE,
    INDEX `teacher_class_assignments_subject_id_foreign` (`subject_id`) USING BTREE,
    CONSTRAINT `teacher_class_assignments_teacher_profile_id_foreign` 
        FOREIGN KEY (`teacher_profile_id`) REFERENCES `teacher_profiles` (`id`) 
        ON UPDATE NO ACTION ON DELETE CASCADE,
    CONSTRAINT `teacher_class_assignments_class_id_foreign` 
        FOREIGN KEY (`class_id`) REFERENCES `classes` (`id`) 
        ON UPDATE NO ACTION ON DELETE CASCADE,
    CONSTRAINT `teacher_class_assignments_subject_id_foreign` 
        FOREIGN KEY (`subject_id`) REFERENCES `subjects` (`id`) 
        ON UPDATE NO ACTION ON DELETE CASCADE
)
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB;
