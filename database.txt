CREATE TABLE `notifications` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `user_id` BIGINT UNSIGNED NOT NULL,
    `type` VARCHAR(255) NOT NULL,
    `title` VARCHAR(255) NOT NULL,
    `message` TEXT NOT NULL,
    `link` VARCHAR(255) NULL,
    `icon` VARCHAR(255) NOT NULL DEFAULT 'fa-bell',
    `color` VARCHAR(255) NOT NULL DEFAULT 'blue',
    `available_at` TIMESTAMP NULL,
    `read_at` TIMESTAMP NULL,
    `created_at` TIMESTAMP NULL,
    `updated_at` TIMESTAMP NULL,

    PRIMARY KEY (`id`),

    INDEX `notifications_user_id_read_at_index` (`user_id`, `read_at`),
    INDEX `notifications_user_id_available_at_index` (`user_id`, `available_at`),

    CONSTRAINT `notifications_user_id_foreign`
        FOREIGN KEY (`user_id`)
        REFERENCES `users` (`id`)
        ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


SELECT id, user_id, title, available_at, NOW() AS current_time
FROM notifications;
